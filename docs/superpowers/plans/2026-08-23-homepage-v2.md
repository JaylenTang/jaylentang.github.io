- [ ] **Step 1: Add one V2 controller**

Append a single inline IIFE. It must:

- Toggle `.is-open` on `#v2-navigation` and keep `aria-expanded` and the button label synchronized.
- Close the menu after an in-page navigation link is selected and when Escape is pressed.
- Reveal the same XOR-protected email used on the current homepage.
- Open original research PNGs from `data-full` values.
- Save the triggering thumbnail in `lastModalTrigger`, move focus to the modal close button, trap Tab between modal controls, and return focus after close.
- Close the modal from its close button, backdrop selection, or Escape.

Use this complete controller so contract assertions and maintenance remain clear:

```js
<script>
  (function () {
    var menuButton = document.querySelector(".v2-menu-toggle");
    var navigation = document.getElementById("v2-navigation");
    var emailTarget = document.getElementById("v2-email");
    var emailButtons = Array.prototype.slice.call(document.querySelectorAll("[data-v2-email], [data-v2-email-icon]"));
    var modal = document.querySelector("[data-v2-modal]");
    var modalImage = modal ? modal.querySelector("img") : null;
    var modalClose = modal ? modal.querySelector(".v2-modal__close") : null;
    var lastModalTrigger = null;

    function setMenuOpen(isOpen) {
      if (!menuButton || !navigation) return;
      navigation.classList.toggle("is-open", isOpen);
      menuButton.setAttribute("aria-expanded", isOpen ? "true" : "false");
      menuButton.setAttribute("aria-label", isOpen ? "Close navigation" : "Open navigation");
    }

    function revealEmail() {
      var encoded = [125, 46, 109, 55, 79, 99, 112, 76, 46, 69, 126, 105, 105, 63, 83];
      var key = [23, 71, 12, 91, 38];
      var email = encoded.map(function (value, index) {
        return String.fromCharCode(value ^ key[index % key.length]);
      }).join("");

      if (emailTarget && !emailTarget.querySelector("a")) {
        var inlineLink = document.createElement("a");
        inlineLink.href = "mailto:" + email;
        inlineLink.textContent = email;
        emailTarget.textContent = "";
        emailTarget.appendChild(inlineLink);
      }

      emailButtons.forEach(function (button) {
        if (button.hasAttribute("data-v2-email")) {
          button.remove();
          return;
        }

        if (!button.isConnected) return;
        var iconLink = document.createElement("a");
        iconLink.className = button.className;
        iconLink.href = "mailto:" + email;
        iconLink.setAttribute("aria-label", "Email Jialin Tang");
        iconLink.setAttribute("title", "Email Jialin Tang");
        iconLink.innerHTML = button.innerHTML;
        button.replaceWith(iconLink);
      });
    }

    function getFocusableModalElements() {
      if (!modal) return [];
      return Array.prototype.slice.call(modal.querySelectorAll("a[href], button:not([disabled]), [tabindex]:not([tabindex='-1'])"));
    }

    function openModal(trigger) {
      if (!modal || !modalImage || !modalClose) return;
      var sourceImage = trigger.querySelector("img");
      lastModalTrigger = trigger;
      modalImage.src = trigger.getAttribute("data-full");
      modalImage.alt = sourceImage ? sourceImage.alt : "Research figure";
      modal.hidden = false;
      document.body.classList.add("v2-modal-open");
      modalClose.focus({ preventScroll: true });
    }

    function closeModal() {
      if (!modal || modal.hidden) return;
      modal.hidden = true;
      modalImage.removeAttribute("src");
      document.body.classList.remove("v2-modal-open");
      if (lastModalTrigger) {
        lastModalTrigger.focus({ preventScroll: true });
      }
    }

    function handleModalKeydown(event) {
      if (!modal || modal.hidden) return;
      if (event.key === "Escape") {
        event.preventDefault();
        closeModal();
        return;
      }
      if (event.key !== "Tab") return;

      var focusable = getFocusableModalElements();
      if (!focusable.length) return;
      var first = focusable[0];
      var last = focusable[focusable.length - 1];

      if (event.shiftKey && document.activeElement === first) {
        event.preventDefault();
        last.focus();
      } else if (!event.shiftKey && document.activeElement === last) {
        event.preventDefault();
        first.focus();
      }
    }

    if (menuButton && navigation) {
      menuButton.addEventListener("click", function () {
        setMenuOpen(menuButton.getAttribute("aria-expanded") !== "true");
      });
      navigation.addEventListener("click", function (event) {
        if (event.target.closest("a[href^='#']")) setMenuOpen(false);
      });
    }

    emailButtons.forEach(function (button) {
      button.addEventListener("click", revealEmail);
    });

    document.querySelectorAll(".v2-publication__figure").forEach(function (trigger) {
      trigger.addEventListener("click", function () {
        openModal(trigger);
      });
    });

    if (modal && modalClose) {
      modalClose.addEventListener("click", closeModal);
      modal.addEventListener("click", function (event) {
        if (event.target === modal) closeModal();
      });
    }

    document.addEventListener("keydown", function (event) {
      if (modal && !modal.hidden) {
        handleModalKeydown(event);
      } else if (event.key === "Escape" && menuButton && menuButton.getAttribute("aria-expanded") === "true") {
        setMenuOpen(false);
        menuButton.focus();
      }
    });
  }());
```
